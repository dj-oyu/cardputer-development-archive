idf_component_register(SRCS "dot8.c" "checked.c" "corpus.c" "control.c" "harness.c" INCLUDE_DIRS "." REQUIRES freertos esp_timer)
if(NOT IDF_TARGET STREQUAL "esp32s3")
    message(FATAL_ERROR "Private PIE validation requires esp32s3")
endif()
set(identity "dot8-control-v2")
foreach(f CMakeLists.txt dot8.c dot8.h checked.c checked.h corpus.c corpus.h control.c control.h harness.c)
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/${f}")
    file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/${f}" h)
    string(APPEND identity "${h}")
endforeach()
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/../sdkconfig.defaults")
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/../sdkconfig.defaults" h)
string(SHA256 identity "${identity}${h}")
string(SUBSTRING "${identity}" 0 16 identity)
target_compile_definitions(${COMPONENT_LIB} PRIVATE DOT8_BUILD_ID="${identity}")
target_compile_options(${COMPONENT_LIB} PRIVATE -std=gnu23 -O2 -ffp-contract=off -fno-lto -fstack-usage -Wall -Wextra -Werror -fno-builtin-memcpy -fno-builtin-memset -fno-builtin-bzero)
set_source_files_properties(dot8.c PROPERTIES COMPILE_OPTIONS "-fno-lto")
