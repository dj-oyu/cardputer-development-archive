29: {
30:   unsigned int fsr;

32:   except <<= _FE_EXCEPTION_FLAGS_OFFSET;
33:   __asm__ volatile ("rur.fsr %0" : "=a"(fsr));
34:   fsr = fsr & ~except;
35:   __asm__ volatile ("wur.fsr %0" : : "a"(fsr));
36:   return 0;
